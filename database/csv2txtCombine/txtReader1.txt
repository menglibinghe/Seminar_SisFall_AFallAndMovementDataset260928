import pandas as pd

file_path = 'Merged_SisFall_Final.txt'

# 1. 读取文件（如果文件超大，可以只读取需要的列以节省内存）
df = pd.read_csv(file_path, sep='\t')

# 2. 筛选 Activity 为 'F15' 的记录
# 注意：如果 Activity 列存的是字符串，用 'F15'；如果是纯数字，请检查是否需要去掉引号
filtered_df = df[df['Activity'] == 'F15']

# 3. 显示前 10 行
print("--- Activity 为 F15 的前 10 行记录 ---")
if not filtered_df.empty:
    print(filtered_df.head(10))
else:
    print("未找到 Activity 为 F15 的记录。请检查列名或数据内容。")
